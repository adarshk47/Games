// AYUSH Vaidya bookings (ayush.mitrojan.in) – Firestore rules block.
//
// The project has ONE set of Firestore rules (shared with Master G). Paste this block into
// firestore_rules.txt, inside `match /databases/{database}/documents { ... }`, just before the
// final `match /{document=**} { allow read, write: if false; }`, then publish the whole file
// in Firebase Console -> Firestore Database -> Rules. If this block is missing from the
// published rules, booking and diet requests stop working and admin.html shows no bookings.
//
// Needs the helper `signedIn()` that firestore_rules.txt already defines.

    // --------------------------------------------------- AYUSH Vaidya bookings
    // ayush_bookings/{YYYY-MM-DD_HHMM_doctor} from ayush.mitrojan.in. One document
    // per doctor + slot: anyone may request a free slot (create only, so a taken
    // slot cannot be overwritten); only clinic admins (ayush.mitrojan.in/admin.html)
    // can read, change the status or delete. Add the clinic's Google accounts below.
    function ayushAdmin() {
      return signedIn() && request.auth.token.email_verified == true
        && request.auth.token.email in ['aiforsmart@gmail.com'];
    }

    function validAyushBooking(id, d) {
      return d.keys().hasOnly(['name', 'phone', 'doctor', 'date', 'time', 'lang', 'reason', 'status', 'consent', 'createdAt'])
        && d.name is string && d.name.size() >= 2 && d.name.size() <= 60
        && d.phone is string && d.phone.matches('^[6-9][0-9]{9}$')
        && d.doctor in ['dolly', 'pooja']
        && d.date is string && d.date.matches('^20[0-9]{2}-[01][0-9]-[0-3][0-9]$')
        && d.time is string && d.time.matches('^[0-2][0-9]:[0-5][0-9]$')
        && id == d.date + '_' + d.time.replace(':', '') + '_' + d.doctor
        && d.lang in ['hi', 'en']
        && (!('reason' in d) || (d.reason is string && d.reason.size() <= 300))
        && d.status == 'requested'
        && d.consent == true
        && d.createdAt == request.time;
    }

    match /ayush_bookings/{id} {
      allow create: if validAyushBooking(id, request.resource.data);
      allow read, delete: if ayushAdmin();
      allow update: if ayushAdmin()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status'])
        && request.resource.data.status in ['requested', 'confirmed', 'done', 'cancelled'];
    }
